0009         function curateModelFromTablesKeepsSupersetReaction(testCase)
0010             % A model reaction that uses the candidate's metabolites *and*
0011             % others is a different reaction, not a duplicate: adding
0012             % A + B -> C to a model holding A + B -> C + H must leave the
0013             % existing reaction alone and add the candidate separately.

0015             [coeffsF, infoF] = tCuration.writeRxnTables(testCase, ...

0017             evalc(['out = curateModelFromTables(model,''none'',''rxnsCoeffs'',' ...
0018                 'coeffsF,''rxnsInfo'',infoF);']);
0019 
0020             testCase.verifyNumElements(out.rxns, 2);
0021             % The existing reaction keeps its name and its fourth metabolite
0022             testCase.verifyEqual(out.rxnNames{1}, 'ATP hydrolysis with proton');

0024             % The candidate is added as a new reaction
