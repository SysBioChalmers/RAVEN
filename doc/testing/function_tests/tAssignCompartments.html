0002 % tAssignCompartments  Tests for the functionality-constrained compartment-assignment MILP.
0003 %
0004 %   Uses a tiny single-compartment toy: EX_A (uptake) -> r1: A->B -> bio: B-> (objective), with
0005 %   gene g1 (on r1) scored higher for a hypothetical compartment 'm' than for the cytosol 'c'.
0006 %   The decisive checks are that the functionality constraint can override the localization score
0007 %   and that the returned model stays able to grow. Requires a MILP solver.

0011         function functionalityOverridesScore(testCase)
0012             testCase.assumeMILPSolver();
0013             model = tAssignCompartments.toy();
0014             GSS = tAssignCompartments.gss();
0015             % B is non-transportable, so placing r1 in 'm' would strand B where biomass (in 'c')
0016             % cannot reach it -> r1 must stay in 'c', against g1's higher 'm' score.
0017             evalc(['[oModel, place, ~, eFlag] = assignCompartments(model, GSS, {''r1''}, ' ...
0018                    '''defaultCompartment'', ''c'', ''transportable'', {}, ''verbose'', false);']);
0019             testCase.verifyEqual(eFlag, 1);
0020             testCase.verifyEqual(place.compartment{1}, 'c');
0021             sol = solveLP(oModel);
0022             testCase.verifyGreaterThan(sol.f, 1e-6);          % applied model still grows
0023         end
