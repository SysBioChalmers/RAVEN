0119         function exportImportSBMLRoundTripPreservesDeltaG(testCase)
0120             % rxnDeltaG/metDeltaG round-trip through a <p>deltaG: VALUE</p>
0121             % notes entry -- the same key and format raven-toolbox reads
0122             % and writes, so either toolbox's SBML file is a complete
0123             % round trip for the other (RAVEN#724). An untouched reaction
0124             % or metabolite stays NaN, and the field is omitted entirely
0125             % if nothing in the model has a value. Values are the kind of
0126             % magnitude a real thermodynamic estimate has, not contrived
0127             % high-precision ones: num2str's default precision (no digits
0128             % argument) is exact for these, same as Confidence Level.
0129             m = testCase.model;
0130             m.rxnDeltaG = NaN(numel(m.rxns), 1);
0131             m.rxnDeltaG(1) = -8.7;
0132             m.metDeltaG = NaN(numel(m.mets), 1);
0133             m.metDeltaG(1) = 7.5;
0134 
0135             f = [tempname '.xml'];
0136             testCase.addTeardown(@() delete(f));
0137             evalc('exportModel(m, f);');
0138             evalc('m2 = importModel(f);');
0139 
0140             testCase.verifyEqual(m2.rxnDeltaG(1), -8.7, 'AbsTol', 1e-9);
0141             testCase.verifyTrue(isnan(m2.rxnDeltaG(2)));
0142             testCase.verifyEqual(m2.metDeltaG(1), 7.5, 'AbsTol', 1e-9);
0143             testCase.verifyTrue(isnan(m2.metDeltaG(2)));
