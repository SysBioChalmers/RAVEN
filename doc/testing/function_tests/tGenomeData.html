0019         function getGeneDataReadsEukaryote(testCase)
0020             % gene -> mRNA -> CDS: the protein accession must come from the
0021             % CDS protein_id (NP_), not the mRNA transcript (NM_), and the
0022             % two CDS lines of one protein collapse to a single row.
0023             T = getGeneData(fullfile(testCase.fixtureDir(),'euk.gff'));
0024             testCase.verifyEqual(height(T), 2);
0025             i1 = find(strcmp(T.locus_tag,'LT1'),1);
