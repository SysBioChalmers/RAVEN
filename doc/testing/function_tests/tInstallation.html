0017         function findRAVENrootSurvivesClearAndRepeatCalls(testCase)
0018             % The resolved path is cached for the session (findRAVENroot is
0019             % called on every solver invocation, so re-reading the
0020             % preference from disk every time is wasteful). Repeated calls
0021             % must keep returning the same root, and it must still resolve

0026             testCase.verifyEqual(p2, p1);
