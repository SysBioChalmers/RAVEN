0015         function guessCompositionAccountsForStoichiometricCoefficient(testCase)
0016             % A -> 2 B, with A's formula known (CH4). B's own coefficient is
0017             % 2, so its atom counts must be half of A's (C0.5H4/2 = C0.5H2),
0018             % not identical to A's.
0019             model = tReconstruction.toyModel();
0020             model.metFormulas = {'CH4'; ''};
0021             evalc('[m2, guessedFor, couldNotGuess] = guessComposition(model);');
0022             testCase.verifyEqual(guessedFor, {'B'});
0023             testCase.verifyEmpty(couldNotGuess);
0024             testCase.verifyEqual(m2.metFormulas{2}, 'C0.5H2');
0025         end
